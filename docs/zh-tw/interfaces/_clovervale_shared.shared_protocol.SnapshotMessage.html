    Interface SnapshotMessage

    The state of every player, and of what there is to pick in the world, which the server sends every peer again and again, with what that peer alone is told of its own player.

    interface SnapshotMessage {
        acquaintances: readonly VillagerId[];
        clovers: readonly number[];
        dialogue: DialogueView | null;
        inventory: readonly (
            | "clover"
            | "sword"
            | "leather-armor"
            | "healing-potion"
            | "clovervale-message"
            | "spy-token"
            | "martha-ointment"
        )[];
        isEndingReached: boolean;
        items: readonly GroundItem[];
        looted: readonly number[];
        monsters: readonly MonsterState[];
        own: InputProgress;
        players: readonly PlayerState[];
        quests: readonly Quest[];
        silver: number;
        slash: SlashJudgment
        | null;
        step: number;
        tobyFind: number | null;
        type: "snapshot";
    }
